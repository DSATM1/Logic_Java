4. Reverse a Number

    rev = rev * 10 + n % 10;
    n /= 10;
}
System.out.println(rev);



5. Palindrome Number
int n = 121;
int temp = n;
int rev = 0;

while (n != 0) {
    rev = rev * 10 + n % 10;
    n /= 10;
}

if (temp == rev)
    System.out.println("Palindrome");
else
    System.out.println("Not Palindrome");

6. Armstrong Number
int n = 153;
int temp = n;
int sum = 0;

while (temp != 0) {
    int digit = temp % 10;
    sum += digit * digit * digit;
    temp /= 10;
}

if (sum == n)
    System.out.println("Armstrong");
else
    System.out.println("Not Armstrong");



7. Prime Number
int n = 29;
boolean prime = true;

if (n <= 1)
    prime = false;

for (int i = 2; i <= Math.sqrt(n); i++) {
    if (n % i == 0) {
        prime = false;
        break;
    }
}

System.out.println(prime ? "Prime" : "Not Prime");

Arrays
Find Largest Element
int[] arr = {10, 45, 23, 89, 12};

int max = arr[0];

for (int i = 1; i < arr.length; i++) {
    if (arr[i] > max)
        max = arr[i];
}

System.out.println(max);


Find Second Largest
int[] arr = {10, 50, 30, 80, 60};

int first = Integer.MIN_VALUE;
int second = Integer.MIN_VALUE;

for (int num : arr) {
    if (num > first) {
        second = first;
        first = num;
    } else if (num > second && num != first) {
        second = num;
    }
}

System.out.println(second);


Reverse an Array
int[] arr = {1,2,3,4,5};

int start = 0;
int end = arr.length - 1;

while (start < end) {
    int temp = arr[start];
    arr[start] = arr[end];
    arr[end] = temp;
    start++;
    end--;
}

for (int num : arr)
    System.out.print(num + " ");


Strings
Reverse a String
String str = "Java";
String rev = "";

for (int i = str.length() - 1; i >= 0; i--) {
    rev += str.charAt(i);
}

System.out.println(rev);


Count Vowels
String str = "Programming";
int count = 0;

str = str.toLowerCase();

for (int i = 0; i < str.length(); i++) {
    char ch = str.charAt(i);

    if (ch == 'a' || ch == 'e' || ch == 'i' ||
        ch == 'o' || ch == 'u') {
        count++;
    }
}

System.out.println(count);
