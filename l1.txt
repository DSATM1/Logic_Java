4. Reverse a Number

int n = 12345;
int rev = 0;

while (n != 0) {
    rev = rev * 10 + n % 10;
    n /= 10;
}
System.out.println(rev);



5. Palindrome Number
int n = 121;
int temp = n;
int rev = 0;

while (n != 0) {
    rev = rev * 10 + n % 10;
    n /= 10;
}

if (temp == rev)
    System.out.println("Palindrome");
else
    System.out.println("Not Palindrome");
