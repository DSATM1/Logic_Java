4. Reverse a Number

int n = 12345;
int rev = 0;

while (n != 0) {
    rev = rev * 10 + n % 10;
    n /= 10;
}
System.out.println(rev);



5. Palindrome Number
int n = 121;
int temp = n;
int rev = 0;

while (n != 0) {
    rev = rev * 10 + n % 10;
    n /= 10;
}

if (temp == rev)
    System.out.println("Palindrome");
else
    System.out.println("Not Palindrome");

6. Armstrong Number
int n = 153;
int temp = n;
int sum = 0;

while (temp != 0) {
    int digit = temp % 10;
    sum += digit * digit * digit;
    temp /= 10;
}

if (sum == n)
    System.out.println("Armstrong");
else
    System.out.println("Not Armstrong");



7. Prime Number
int n = 29;
boolean prime = true;

if (n <= 1)
    prime = false;

for (int i = 2; i <= Math.sqrt(n); i++) {
    if (n % i == 0) {
        prime = false;
        break;
    }
}

System.out.println(prime ? "Prime" : "Not Prime");

8. Factorial
int n = 5;
long fact = 1;

for (int i = 1; i <= n; i++) {

}

System.out.println(fact);


9. Fibonacci Series
int n = 10;
int a = 0, b = 1;

for (int i = 1; i <= n; i++) {
    System.out.print(a + " ");
    int c = a + b;
    a = b;
    b = c;
}



10. Count Digits

    count++;
